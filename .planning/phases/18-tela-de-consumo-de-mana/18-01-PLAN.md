Create `src/features/mana/hooks/useManaStatus.ts` following the useUserTier pattern:

```typescript
'use client'

import { useQuery } from '@tanstack/react-query'
import { manaApi } from '../services/manaApi'
import { ManaStatus, ManaHistoryDay } from '../types/mana.types'
import { tokenStorage } from '@/shared/lib/apiClient'

const MANA_STATUS_QUERY_KEY = ['mana-status']

// TODO: Replace with real endpoint when backend supports GET /api/v1/mana/history (per D-04)
function getMockHistory(currentMana: number, maxMana: number): ManaHistoryDay[] {
  const days = ['Seg', 'Ter', 'Qua', 'Qui', 'Sex', 'Sab', 'Dom']
  return days.map((day) => ({
    day,
    used: Math.floor(Math.random() * maxMana * 0.8),
  }))
}

export function useManaStatus() {
  const isAuthenticated = typeof window !== 'undefined' && !!tokenStorage.getAccessToken()

  const { data, isLoading, error } = useQuery<ManaStatus>({
    queryKey: MANA_STATUS_QUERY_KEY,
    queryFn: () => manaApi.getStatus(),
    staleTime: 2 * 60 * 1000, // 2 minutes (mana changes more frequently than tier)
    retry: false,
    enabled: isAuthenticated,
  })

  const currentMana = data?.currentMana ?? 0
  const maxMana = data?.maxMana ?? 100
  const percentage = maxMana > 0 ? Math.round((currentMana / maxMana) * 100) : 0
  const nextResetAt = data?.nextResetAt ?? null
  const tier = data?.tier ?? 'free'
  const history = data ? getMockHistory(currentMana, maxMana) : []

  return {
    currentMana,
    maxMana,
    percentage,
    nextResetAt,
    tier,
    history,
    isLoading,
    error,
  }
}
```
